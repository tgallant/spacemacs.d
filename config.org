      ;; only check on save
      (setq flycheck-check-syntax-automatically '(mode-enabled save))

      ;; From https://github.com/flycheck/flycheck/issues/1974#issuecomment-1343495202
      (flycheck-define-checker python-ruff
        "A Python syntax and style checker using the ruff utility.
      To override the path to the ruff executable, set
      `flycheck-python-ruff-executable'.
      See URL `http://pypi.python.org/pypi/ruff'."
        :command ("ruff"
                  "check"
                  "--output-format=full"
                  (eval (when buffer-file-name
                          (concat "--stdin-filename=" buffer-file-name)))
                  "-")

        :error-filter (lambda (errors)
                        (let ((errors (flycheck-sanitize-errors errors)))
                          (seq-map #'flycheck-flake8-fix-error-level errors)))

        ((warning line-start
                  (file-name) ":" line ":" (optional column ":") " "
                  (id (one-or-more (any alpha)) (one-or-more digit)) " "
                  (message (one-or-more not-newline))
                  line-end))
        :modes python-mode)

      (add-hook 'python-mode-hook
                (lambda ()
                  (flycheck-mode 1)
                  (semantic-mode 1)
                  (setq flycheck-checker 'python-ruff

*** start new worktree
    #+begin_src emacs-lisp
      (defun current-project ()
        (-> (projectile-project-root)
            (substring 0 -1)))

      (defun construct-worktree-path (name)
        (string-join (list (current-project) "-" name)))

      (defun make-wt (name dir)
        (list name dir))

      (defun wt-name (wt)
        (nth 0 wt))

      (defun wt-dir (wt)
        (nth 1 wt))

      (defun create-worktree (name)
        (let ((dir (construct-worktree-path name)))
          (magit-worktree-branch dir name (magit-get-current-branch))
          (make-wt name dir)))

      (defun switch-project (wt)
        (projectile-switch-project-by-name (wt-dir wt))
        wt)

      (defun start-tmux-window (wt)
        (shell-command
         (format "tmux new-window -c %s -n %s 'gemini --yolo --skip-trust'"
                 (shell-quote-argument (wt-dir wt))
                 (shell-quote-argument (wt-name wt)))))

      (defun my/create-tmux-window ()
        (interactive)
        (-> (magit-get-current-branch)
            (make-wt default-directory)
            (start-tmux-window)))

      (defun my/new-worktree-session ()
        (interactive)
        (-> (read-string "New worktree name: ")
            (create-worktree)
            (switch-project)
            (start-tmux-window)))

      (spacemacs/set-leader-keys "on" 'my/new-worktree-session)

      (spacemacs/set-leader-keys "oc" 'my/create-tmux-window)

*** Replay commits on top of branch
    Using sleep-for works fine but would be good to figure out a more reliable

#+begin_src emacs-lisp
  (defun wait-for-git-rebase ()
    (print "wait for rebase")
    (if (not (file-exists-p ".git/REBASE_HEAD"))
        (progn
          (sleep-for 1)
          (print "checking for rebase...")
          (wait-for-git-rebase))))

  (defun wait-for-git-lock ()
    (print "wait for lock")
    (if (not (file-exists-p ".git/index.lock"))
        (progn
          (sleep-for 1)
          (print "checking for index lock...")
          (wait-for-git-lock))))

  (defun wait-for-git-lock-release ()
    (print "wait for lock release")
    (if (file-exists-p ".git/index.lock")
        (progn
          (sleep-for 1)
          (print "checking for index lock release...")
          (wait-for-git-lock-release))))

  (defun replay-commits-on-branch (commit)
    "Replay the commits starting at the point on top of another branch"
    (interactive (list (magit-reset-read-branch-or-commit
                        (concat (magit--propertize-face "Hard" 'bold)
                                " reset %s to"))))
    (let ((commit-at-point (magit-commit-at-point)))
      (magit-rebase-edit-commit commit-at-point '())
      (sleep-for 1)
      (magit-reset-hard commit)
      (sleep-for 1)
      (magit-rebase-continue)))

  (with-eval-after-load 'magit
    (transient-append-suffix 'magit-rebase "s"
      '("b" "Replay commits on branch" replay-commits-on-branch)))

*** Add and push tag
#+begin_src emacs-lisp
  (defun today ()
    (format-time-string "%Y-%m-%d"))

  (defun tag-base (prefix)
    (format "%s-%s-tim" prefix (today)))

  (defun tag-version (tag version)
    (format "%s-v%d" tag version))

  (defun next-tag-version (tag)
    (cl-loop for v from 1 to 100
             with next-tag
             with version
             do (setq version (tag-version tag v))
             if (not (magit-rev-verify version)) return version))

  (defun create-tag (tag)
    (magit-call-git "tag" tag "HEAD")
    tag)

  (defun push-tag (tag)
    (magit-git-push "origin" tag nil))

  (defun tag-and-push (prefix)
    (-> (tag-base prefix)
        (next-tag-version)
        (create-tag)
        (push-tag)))

  (defun my/tag-and-push-dev ()
    (interactive)
    (tag-and-push "dev-all"))

  (defun my/tag-and-push-stg ()
    (interactive)
    (tag-and-push "feature-stg"))

  (spacemacs/declare-prefix "og" "git")

  (spacemacs/declare-prefix "ogt" "tags")

  (spacemacs/set-leader-keys "ogtd" 'my/tag-and-push-dev)

  (spacemacs/set-leader-keys "ogts" 'my/tag-and-push-stg)
